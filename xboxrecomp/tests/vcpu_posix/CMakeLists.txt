cmake_minimum_required(VERSION 3.20)
project(vcpu_posix_test C)
if(WIN32)
    message(FATAL_ERROR "This regression covers POSIX guest preemption")

set(CMAKE_C_STANDARD 11)
add_subdirectory(../../src/platform platform)

enable_testing()

set_tests_properties(vcpu_posix_test PROPERTIES TIMEOUT 20)
