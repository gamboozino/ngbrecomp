cmake_minimum_required(VERSION 3.20)

# ============================================================================
# Ninja Gaiden Black - static recompilation (xboxrecomp)
#
# Native Linux (x86-64), Windows and macOS (arm64) builds. The generated code comes from the
# xboxrecomp pipeline (see README.md); it is not checked in.
# ============================================================================

project(ninja_gaiden_black C CXX)
if(APPLE AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    message(FATAL_ERROR "The macOS port currently supports Apple Silicon only")
endif()
if(WIN32)
    enable_language(RC)   # the program's icon (packaging/windows/ngb.rc)
endif()   # CXX: the GPU renderer links VMA

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)

# Bring-up build: record every indirect-branch target the title reaches, so
# tools/feedback.sh can reseed function detection with them. Applies to the
# toolkit too (its icall_feedback.c compiles to nothing without it).
option(NGB_ICALL_FEEDBACK "Record indirect-branch targets (bring-up builds)" OFF)
if(NGB_ICALL_FEEDBACK)
    add_compile_definitions(RECOMP_ICALL_FEEDBACK)
endif()

# The GPU: xemu's NV2A model with its Vulkan renderer (xboxrecomp nv2a_vk).
option(NGB_NV2A_VK "Render with the NV2A Vulkan model" ON)
set(XBOXRECOMP_NV2A_VK ${NGB_NV2A_VK} CACHE BOOL "" FORCE)
set(XBOXRECOMP_SDL_BACKENDS ON CACHE BOOL "" FORCE)   # Windows: same SDL audio/input as Linux
set(XBOXRECOMP_WIN_VCPU ON CACHE BOOL "" FORCE)        # Windows: one virtual CPU, as on Linux

set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/xboxrecomp" CACHE PATH
    "Path to the xboxrecomp toolkit root directory")
add_subdirectory(${XBOXRECOMP_DIR} ${CMAKE_BINARY_DIR}/xboxrecomp)

# ── Recompiled code ─────────────────────────────────────────────

# CONFIGURE_DEPENDS: a regenerated tree can add or drop chunks.
file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS "src/recomp/gen/*.c")
if(NOT RECOMP_GEN_SOURCES)
    message(FATAL_ERROR "src/recomp/gen is empty: run the recomp pipeline first (README.md)")
endif()

# The generated chunks are ~2 MB of C each. They are an object library so
# their flags can differ from the hand-written code, and so a host change
# never recompiles them.
# Fonts (SIL Open Font License, assets/fonts) built into the program: the
# pause menu's pages until the game's own letters are loaded, its key labels
# (src/pause_tab.c) and the frame rate overlay (src/fps_text.c).
set(NGB_FONTS
    "EBGaramond-ExtraBoldItalic.ttf\;ngb_font_tab"
    "LiberationSerif-Regular.ttf\;ngb_font_body"
    "LiberationSerif-Bold.ttf\;ngb_font_label")
set(NGB_FONT_SOURCES)
foreach(entry ${NGB_FONTS})
    list(GET entry 0 file)
    list(GET entry 1 name)
    set(out ${CMAKE_BINARY_DIR}/fonts/${name}.c)
    add_custom_command(OUTPUT ${out}
        COMMAND ${CMAKE_COMMAND} -DIN=${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/${file}
                -DOUT=${out} -DNAME=${name} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/${file}
                ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake
        VERBATIM)
    list(APPEND NGB_FONT_SOURCES ${out})
endforeach()

# The keyboard and mouse prompts (Xelu's free prompts, CC0: assets/prompts/License-Xelu.txt), the dark
# set and its blank key: each embedded, and a table of them by file name (src/pause_tab.c).
file(GLOB NGB_KEY_PNGS ${CMAKE_CURRENT_SOURCE_DIR}/assets/prompts/keys/*.png)
set(key_decls "")
set(key_rows "")
list(LENGTH NGB_KEY_PNGS key_count)
foreach(png ${NGB_KEY_PNGS})
    get_filename_component(stem ${png} NAME_WE)
    string(MAKE_C_IDENTIFIER "ngb_key_${stem}" sym)
    set(out ${CMAKE_BINARY_DIR}/keys/${sym}.c)
    add_custom_command(OUTPUT ${out}
        COMMAND ${CMAKE_COMMAND} -DIN=${png} -DOUT=${out} -DNAME=${sym}
                -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake
        DEPENDS ${png} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake
        VERBATIM)
    list(APPEND NGB_FONT_SOURCES ${out})
    string(APPEND key_decls "extern const unsigned char ${sym}[]; extern const unsigned int ${sym}_len;\n")
    string(APPEND key_rows "    { \"${stem}\", ${sym}, &${sym}_len },\n")
endforeach()
file(WRITE ${CMAKE_BINARY_DIR}/keys/key_table.c.in
     "/* Generated by CMakeLists.txt: the embedded key prompts by name. */\n${key_decls}"
     "const struct { const char *name; const unsigned char *data; const unsigned int *len; } ngb_key_pngs[] = {\n"
     "${key_rows}};\nconst int ngb_key_png_count = ${key_count};\n")
configure_file(${CMAKE_BINARY_DIR}/keys/key_table.c.in ${CMAKE_BINARY_DIR}/keys/key_table.c COPYONLY)
list(APPEND NGB_FONT_SOURCES ${CMAKE_BINARY_DIR}/keys/key_table.c)

add_library(ngb_gen OBJECT ${RECOMP_GEN_SOURCES}
    # Hand-written replacements in the generated code's style.
    ${CMAKE_CURRENT_SOURCE_DIR}/src/movie_skip.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/costume_menu.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/pc_menu.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/pause_tab.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/menu_mouse.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/menu_font.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/fps_text.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/playtime.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/main_menu.c
    # The Direct3D hooks (docs/renderer.md, tools/take_d3d.py).
    ${CMAKE_CURRENT_SOURCE_DIR}/src/d3d/d3d.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/cheats.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/d3d/d3d_hooks.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/d3d/d3d_orig.c
    ${NGB_FONT_SOURCES})
target_include_directories(ngb_gen PRIVATE
    ${XBOXRECOMP_DIR}/third_party/stb
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen
    ${XBOXRECOMP_DIR}/include
    ${XBOXRECOMP_DIR}/src
    ${XBOXRECOMP_DIR}/src/kernel
)
target_link_libraries(ngb_gen PRIVATE platform)
if(NGB_NV2A_VK)
    find_package(SDL2 REQUIRED)   # SDL's key codes (src/pause_tab.c's key prompts)
    target_include_directories(ngb_gen PRIVATE $<TARGET_PROPERTY:SDL2::SDL2,INTERFACE_INCLUDE_DIRECTORIES>)
endif()

if(MSVC)
    target_compile_options(ngb_gen PRIVATE /bigobj /O2)
else()
    # The guest memory model aliases freely through MEM32/MEMF; strict
    # aliasing would let the optimiser reorder guest loads and stores.
    # Lifted code is one big function per guest routine, so the flags that
    # make a function's codegen superlinear are capped.
    target_compile_options(ngb_gen PRIVATE
        -O2 -fno-strict-aliasing -fwrapv
        -Wno-unused-label -Wno-unused-variable -Wno-unused-but-set-variable
        -Wno-parentheses-equality -Wno-tautological-compare
        -Wno-incompatible-pointer-types-discards-qualifiers
    )
endif()

# ── The Direct3D renderer (docs/renderer.md, src/d3d/NOTICE) ────

if(NGB_NV2A_VK)
    if(APPLE)
        find_package(glslang CONFIG REQUIRED)
    endif()
    set(XRTP ${XBOXRECOMP_DIR}/third_party)
    add_library(ngb_d3d OBJECT
        src/d3d/hle.cpp
        src/d3d/vk/backend.cpp
        src/d3d/vk/spirv.cpp
        src/d3d/gpu/shaders.cpp
        src/d3d/gpu/vertex.cpp
        src/d3d/gpu/texture.cpp)
    target_include_directories(ngb_d3d PRIVATE
        src
        src/d3d src/d3d/compat ${XBOXRECOMP_DIR}/src/nv2a_vk/host
        ${XRTP}/Vulkan-Headers/include ${XRTP}/volk ${XRTP}/VulkanMemoryAllocator/include)
    target_compile_definitions(ngb_d3d PRIVATE VK_NO_PROTOTYPES
        VMA_STATIC_VULKAN_FUNCTIONS=0 VMA_DYNAMIC_VULKAN_FUNCTIONS=0)
    if(APPLE)
        target_link_libraries(ngb_d3d PRIVATE glslang::glslang glslang::glslang-default-resource-limits)
    endif()
    target_compile_features(ngb_d3d PRIVATE cxx_std_20)
    target_compile_options(ngb_d3d PRIVATE -O2 -fno-strict-aliasing)
endif()

# ── Executable ──────────────────────────────────────────────────

add_executable(${PROJECT_NAME}
    src/main.c
    src/recomp_manual.c
    src/fibers.c
    $<$<BOOL:${NGB_NV2A_VK}>:${CMAKE_CURRENT_SOURCE_DIR}/src/launcher.c>
    $<$<BOOL:${NGB_NV2A_VK}>:${CMAKE_CURRENT_SOURCE_DIR}/src/ingame_menu.c>
    $<$<BOOL:${NGB_NV2A_VK}>:${CMAKE_CURRENT_SOURCE_DIR}/src/keyboard.c>
    $<$<BOOL:${NGB_NV2A_VK}>:${CMAKE_CURRENT_SOURCE_DIR}/src/bindings.c>
    src/install.c
    $<TARGET_OBJECTS:ngb_gen>
    $<$<BOOL:${NGB_NV2A_VK}>:$<TARGET_OBJECTS:ngb_d3d>>
    $<$<PLATFORM_ID:Windows>:${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/ngb.rc>
)
target_link_libraries(${PROJECT_NAME} PRIVATE xboxrecomp)
if(WIN32)   # the resource script includes these: rebuilt when they change, not only ngb.rc
    set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/ngb.rc PROPERTIES OBJECT_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/ngb.manifest;${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/ngb.ico")
endif()
if(NGB_NV2A_VK)
    target_link_libraries(${PROJECT_NAME} PRIVATE xbox_nv2a_vk)
    target_compile_definitions(${PROJECT_NAME} PRIVATE NGB_NV2A_VK)
endif()
target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen
    ${XBOXRECOMP_DIR}/include
    ${XBOXRECOMP_DIR}/src
)

if(WIN32)
    # A GUI program (no console window); main() attaches to a parent console.
    set_target_properties(${PROJECT_NAME} PROPERTIES OUTPUT_NAME NinjaGaidenBlack
                          WIN32_EXECUTABLE TRUE)
    target_link_libraries(${PROJECT_NAME} PRIVATE
        winmm dbghelp comdlg32 comctl32 shell32 ole32)
    if(MSVC)
        target_link_options(${PROJECT_NAME} PRIVATE /LARGEADDRESSAWARE)
    endif()
else()
    set_target_properties(${PROJECT_NAME} PROPERTIES OUTPUT_NAME NinjaGaidenBlack)
    # -rdynamic: the crash handler names the faulting sub_XXXXXXXX via dladdr.
    target_link_options(${PROJECT_NAME} PRIVATE -rdynamic)
    target_link_libraries(${PROJECT_NAME} PRIVATE ${CMAKE_DL_LIBS} m)
endif()

# ── First-run setup window (Qt) ─────────────────────────────────
# ngb-setup asks for the player's disc and installs it (src/install.c); the
# game runs it from beside itself when it finds no game files. Optional:
# without Qt the game asks through zenity/kdialog.
if(NOT WIN32)
    find_package(Qt6 QUIET COMPONENTS Widgets)
    if(Qt6_FOUND)
        set(CMAKE_AUTORCC ON)
        add_executable(ngb-setup src/setup/ngb_setup.cpp src/setup/setup.qrc src/install.c)
        target_link_libraries(ngb-setup PRIVATE Qt6::Widgets)
        set_target_properties(ngb-setup PROPERTIES CXX_STANDARD 17
            RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
        message(STATUS "ngb-setup: Qt ${Qt6_VERSION}")
    else()
        message(STATUS "ngb-setup: no Qt6 Widgets, not built")
    endif()
endif()
